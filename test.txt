const About = () => {
  const longText =
    "Ini adalah contoh pertama. Komponen pembungkus sekarang bisa digunakan ulang dengan sangat mudah. Dan ini adalah contoh kedua di bagian bawah halaman. Anda bisa menaruh gambar, video, atau komponen lain di dalamnya!";

  const sectionRef = useRef<HTMLDivElement>(null);
  const [scrollY, setScrollY] = useState(0);

  useEffect(() => {
    const onScroll = () => setScrollY(window.scrollY);
    window.addEventListener("scroll", onScroll, { passive: true });
    return () => window.removeEventListener("scroll", onScroll);
  }, []);

  const sectionTop = sectionRef.current?.offsetTop ?? 0;
  const relativeScroll = scrollY - sectionTop;

  return (
    // Section luar: area scroll panjang, position relative sebagai anchor gambar
    <div ref={sectionRef} className="relative h-[500vh] w-full overflow-hidden">
      {/* TEKS: sticky di tengah layar */}
      <div className="sticky top-0 h-screen flex items-center justify-center z-10 pointer-events-none">
        <StickyScrollContainer
          parentHeight="h-screen"
          containerClassName="w-full max-w-[900px] h-screen flex items-center justify-center"
        >
          {(progress) => (
            <div className="w-[85%] md:w-[70%]">
              <ScrollHighlightText text={longText} progress={progress} />
            </div>
          )}
        </StickyScrollContainer>
      </div>

      {/* GAMBAR: absolute di dalam section, parallax via translateY */}
      {IMAGES.map((img, i) => {
        const parallaxY = relativeScroll * img.speed;
        const style: React.CSSProperties = {
          position: "absolute",
          top: img.top,
          transform: `translateY(-${parallaxY}px)`,
          willChange: "transform",
          zIndex: 5,
          ...(img.left ? { left: img.left } : {}),
          ...(img.right ? { right: img.right } : {}),
        };

        return (
          <div key={i} style={style}>
            <img
              src={img.src}
              alt={img.alt}
              className={`${img.w} ${img.h} object-cover rounded-2xl shadow-2xl`}
            />
          </div>
        );
      })}
    </div>
  );
};

export default About;
